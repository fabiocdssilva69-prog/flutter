import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:go_router/go_router.dart';
import '../../../core/theme/app_colors.dart';
import '../../../core/utils/async_value_ui.dart';
import '../../../domain/entities/enums.dart';
import '../controllers/vacancy_controller.dart';

/// Tela de criação de vagas para barbearias
/// 
/// Features:
/// - Formulário completo para criar vagas
/// - Validação de campos obrigatórios
/// - Campo de comissão condicional (apenas para tipo commission)
/// - Integração com [`VacancyController`]
/// - Feedback visual de loading e erros
/// 
/// Exemplo de uso no router:
/// ```dart
/// GoRoute(
///   path: '/create-vacancy',
///   builder: (context, state) => const CreateVacancyScreen(),
/// )
/// ```
class CreateVacancyScreen extends ConsumerStatefulWidget {
  const CreateVacancyScreen({super.key});

  @override
  ConsumerState<CreateVacancyScreen> createState() =>
      _CreateVacancyScreenState();
}

class _CreateVacancyScreenState extends ConsumerState<CreateVacancyScreen> {
  final _formKey = GlobalKey<FormState>();
  final _titleController = TextEditingController();
  final _workHoursController = TextEditingController();
  final _commissionController = TextEditingController();
  final _requirementsController = TextEditingController();

  VacancyType? _selectedType;

  @override
  void dispose() {
    _titleController.dispose();
    _workHoursController.dispose();
    _commissionController.dispose();
    _requirementsController.dispose();
    super.dispose();
  }

  /// Valida e submete o formulário
  Future<void> _submit() async {
    // Validação de tipo de vaga
    if (_selectedType == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("Selecione o tipo de vaga.")),
      );
      return;
    }

    // Validação do formulário
    if (!_formKey.currentState!.validate()) {
      return;
    }

    // Validação específica para Comissão
    if (_selectedType == VacancyType.commission) {
      if (_commissionController.text.isEmpty) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text("Informe a porcentagem da comissão."),
          ),
        );
        return;
      }

      final commission = double.tryParse(_commissionController.text);
      if (commission == null || commission <= 0 || commission > 100) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text("Informe uma porcentagem válida entre 1 e 100."),
          ),
        );
        return;
      }
    }

    // Converte comissão para double (se aplicável)
    final commission = _selectedType == VacancyType.commission
        ? double.tryParse(_commissionController.text)
        : null;

    // Chama o controller para criar a vaga
    final success = await ref.read(vacancyControllerProvider.notifier).createVacancy(
          title: _titleController.text.trim(),
          type: _selectedType!,
          workHours: _workHoursController.text.trim(),
          commissionPercentage: commission,
          requirements: _requirementsController.text.trim().isEmpty
              ? null
              : _requirementsController.text.trim(),
        );

    if (success && mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text("Vaga criada com sucesso!"),
          backgroundColor: Colors.green,
        ),
      );
      context.pop(); // Fecha a tela de criação
    }
  }

  @override
  Widget build(BuildContext context) {
    // Listener para mostrar erros do controller
    ref.listen<AsyncValue<void>>(
      vacancyControllerProvider,
      (_, state) => state.showAlertDialogOnError(context),
    );

    final state = ref.watch(vacancyControllerProvider);
    final isLoading = state.isLoading;

    return Scaffold(
      appBar: AppBar(
        title: const Text("Publicar Nova Vaga"),
        elevation: 0,
      ),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(16.0),
        child: Form(
          key: _formKey,
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              // Header informativo
              Card(
                color: AppColors.primary.withOpacity(0.1),
                child: const Padding(
                  padding: EdgeInsets.all(16.0),
                  child: Row(
                    children: [
                      Icon(Icons.info_outline, color: AppColors.primary),
                      SizedBox(width: 12),
                      Expanded(
                        child: Text(
                          'Publique sua vaga e encontre profissionais qualificados',
                          style: TextStyle(fontSize: 14),
                        ),
                      ),
                    ],
                  ),
                ),
              ),
              const SizedBox(height: 24),

              // Título da Vaga
              TextFormField(
                controller: _titleController,
                decoration: const InputDecoration(
                  labelText: "Título da Vaga",
                  hintText: "Ex: Barbeiro Profissional - Centro",
                  prefixIcon: Icon(Icons.work_outline),
                  border: OutlineInputBorder(),
                  helperText: "Seja específico e atrativo",
                ),
                enabled: !isLoading,
                textCapitalization: TextCapitalization.words,
                validator: (v) =>
                    (v == null || v.trim().isEmpty) ? "Campo obrigatório" : null,
              ),
              const SizedBox(height: 16),

              // Tipo de Vaga (Dropdown)
              DropdownButtonFormField<VacancyType>(
                decoration: const InputDecoration(
                  labelText: "Tipo de Contratação",
                  prefixIcon: Icon(Icons.category_outlined),
                  border: OutlineInputBorder(),
                  helperText: "Selecione o modelo de trabalho",
                ),
                value: _selectedType,
                items: VacancyType.values.map((type) {
                  String label;
                  IconData icon;
                  switch (type) {
                    case VacancyType.freelancer:
                      label = "Freelancer";
                      icon = Icons.person_outline;
                      break;
                    case VacancyType.clt:
                      label = "CLT (Fixo)";
                      icon = Icons.badge_outlined;
                      break;
                    case VacancyType.commission:
                      label = "Comissionado (%)";
                      icon = Icons.percent;
                      break;
                  }
                  return DropdownMenuItem(
                    value: type,
                    child: Row(
                      children: [
                        Icon(icon, size: 20, color: AppColors.primary),
                        const SizedBox(width: 8),
                        Text(label),
                      ],
                    ),
                  );
                }).toList(),
                onChanged: isLoading
                    ? null
                    : (value) {
                        setState(() {
                          _selectedType = value;
                          // Limpa comissão se não for tipo commission
                          if (value != VacancyType.commission) {
                            _commissionController.clear();
                          }
                        });
                      },
                validator: (v) => v == null ? "Selecione um tipo" : null,
              ),
              const SizedBox(height: 16),

              // Campo de Comissão (Condicional)
              if (_selectedType == VacancyType.commission) ...[
                TextFormField(
                  controller: _commissionController,
                  decoration: const InputDecoration(
                    labelText: "Porcentagem da Comissão",
                    hintText: "Ex: 60",
                    prefixIcon: Icon(Icons.attach_money),
                    suffixText: "%",
                    border: OutlineInputBorder(),
                    helperText: "Valor entre 1 e 100",
                  ),
                  keyboardType: TextInputType.number,
                  enabled: !isLoading,
                  validator: (v) {
                    if (v == null || v.trim().isEmpty) {
                      return "Campo obrigatório para vagas comissionadas";
                    }
                    final value = double.tryParse(v);
                    if (value == null || value <= 0 || value > 100) {
                      return "Informe um valor válido (1-100)";
                    }
                    return null;
                  },
                ),
                const SizedBox(height: 16),
              ],

              // Horário de Trabalho
              TextFormField(
                controller: _workHoursController,
                decoration: const InputDecoration(
                  labelText: "Horário de Trabalho",
                  hintText: "Ex: Segunda a Sexta, 9h às 18h",
                  prefixIcon: Icon(Icons.schedule),
                  border: OutlineInputBorder(),
                  helperText: "Especifique dias e horários",
                ),
                enabled: !isLoading,
                textCapitalization: TextCapitalization.sentences,
                maxLines: 2,
                validator: (v) =>
                    (v == null || v.trim().isEmpty) ? "Campo obrigatório" : null,
              ),
              const SizedBox(height: 16),

              // Requisitos (Opcional)
              TextFormField(
                controller: _requirementsController,
                decoration: const InputDecoration(
                  labelText: "Requisitos e Experiências (Opcional)",
                  hintText:
                      "Ex: Experiência mínima de 2 anos, especialização em degradê...",
                  prefixIcon: Icon(Icons.checklist),
                  border: OutlineInputBorder(),
                  helperText: "Descreva o perfil ideal do candidato",
                ),
                enabled: !isLoading,
                textCapitalization: TextCapitalization.sentences,
                maxLines: 4,
              ),
              const SizedBox(height: 32),

              // Botão de Publicar
              ElevatedButton.icon(
                onPressed: isLoading ? null : _submit,
                icon: isLoading
                    ? const SizedBox(
                        width: 20,
                        height: 20,
                        child: CircularProgressIndicator(
                          strokeWidth: 2,
                          color: AppColors.background,
                        ),
                      )
                    : const Icon(Icons.publish),
                label: Text(isLoading ? "PUBLICANDO..." : "PUBLICAR VAGA"),
                style: ElevatedButton.styleFrom(
                  minimumSize: const Size(double.infinity, 56),
                  backgroundColor: AppColors.primary,
                  foregroundColor: AppColors.background,
                  shape: RoundedRectangleBorder(
                    borderRadius: BorderRadius.circular(12),
                  ),
                ),
              ),
              const SizedBox(height: 12),

              // Botão de Cancelar
              TextButton(
                onPressed: isLoading ? null : () => context.pop(),
                child: const Text("Cancelar"),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
